Overview

This vignette illustrates the competing-risks workflow on the aGVHD dataset: build a task, fit registered engines, obtain tidy predictions/metrics, and visualize with standard verbs.

1. Packages and data

First, we load the necessary libraries and the dataset. We will use the BMT competing risks dataset for this analysis.

library(ml4time2event)
library(dplyr)
library(ggplot2)
library(recipes)

# Load the BMT competing risks data (using this as a proxy for aGVHD analysis)
full_data <- get_bmt_competing_risks_data()

# Take a sample for this vignette to run faster
set.seed(123)
sample_indices <- sample(nrow(full_data), size = min(500, nrow(full_data)))
data_raw <- full_data[sample_indices, ]

## Choose event of interest (1 = transplant-related mortality, 2 = relapse)
event_of_interest <- 1L

# We'll create binary `status` and `cause` columns after splitting so we can
# ensure the training split contains the chosen event. If not, we'll pick an
# available non-zero event automatically.

Create a simple hold-out split for validation.

set.seed(123)
# remove rows of data with missing in target variables
data_raw <- data_raw |>
  filter(!is.na(ftime), !is.na(status))

# Convert multi-class status into binary status and a separate cause column
# status: 0=censored, 1=TRM, 2=relapse
# For competing risks: status becomes binary (0=censored, 1=any event), cause captures the original event type
data <- data_raw |>
  mutate(
    cause = if_else(status == 0L, NA_integer_, as.integer(status)),
    status = if_else(status == 0L, 0L, 1L)
  )

# Now split the transformed data
train_rows <- sample.int(nrow(data), size = floor(0.7 * nrow(data)))
data_train <- data[train_rows, ]
data_test <- data[-train_rows, ]

# Impute missing values in predictors using recipes
# Exclude outcome columns from imputation
exclude_cols <- c("ftime", "status", "cause")
predictor_cols <- setdiff(colnames(data_train), exclude_cols)

# Create recipe for imputation
impute_recipe <- recipe(~., data = data_train) |>
  update_role(all_of(exclude_cols), new_role = "outcome") |>
  step_impute_median(all_numeric_predictors()) |>
  step_impute_mode(all_nominal_predictors())

# Fit recipe on training data
impute_recipe_prep <- prep(impute_recipe, training = data_train)

# Apply to both training and test data
data_train <- bake(impute_recipe_prep, new_data = data_train)
data_test <- bake(impute_recipe_prep, new_data = data_test)

sprintf(
  "Training rows: %s  |  Validation rows: %s",
  nrow(data_train), nrow(data_test)
)

2. Build a competing-risk task

# Create the competing risks task
t2e_task <- ml4t2e_task_cr(
  data = data_train,
  time = "ftime",
  status = "status",
  cause = "cause",
  features = setdiff(colnames(data_train), c("ftime", "status", "cause"))
)

4. Score a hold-out set

# Create task for test data
t2e_task_test <- ml4t2e_task_cr(
  data = data_test,
  time = "ftime",
  status = "status",
  cause = "cause",
  features = setdiff(colnames(data_test), c("ftime", "status", "cause"))
)

# Get predictions on test set
cr_preds_val <- predict(
  cr_fit,
  newdata = data_test,
  times = seq(0, max(data_test$ftime, na.rm = TRUE), length.out = 60),
  type = "cif",
  include = c("ensemble", "cox", "fine_gray", "cr_xgboost")
)

ml4t2e_evaluate(
  cr_preds_val,
  task = t2e_task_test,
  metrics = c("c_index", "ibs")

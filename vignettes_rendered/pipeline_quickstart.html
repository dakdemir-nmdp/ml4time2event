Pipeline Quickstart

Survival pipelines

lung <- get_lung_survival_data() |>
  mutate(status = if_else(status == 2, 1L, status))

surv_pipe <- ml4t2e_pipeline(
  outcome = list(type = "survival", time = "time", event = "status"),
  models = c("cox", "gbm"),
  ensemble = "auto",
  recipe = recipe(~., data = lung) |>
    step_impute_median(all_numeric_predictors()),
  resampling = vfold_cv(lung, v = 3)
)

surv_pipe$fit(lung)

Competing-risk pipelines

bmt <- get_bmt_competing_risks_data() |>
  mutate(
    cause = if_else(status == 0L, NA_integer_, status),
    status = if_else(status == 0L, 0L, 1L)
  )

cr_pipe <- ml4t2e_pipeline(
  outcome = list(
    type = "competing_risk",
    time = "ftime",
    status = "status",
    cause = "cause"
  ),
  models = c("cox", "cr_fine_gray"),
  ensemble = "auto"
)

cr_pipe$fit(bmt)
cr_pipe$summary()
